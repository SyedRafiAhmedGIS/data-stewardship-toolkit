04 Usage (copy & paste)

git clone https://github.com/YOUR-USERNAME/data-stewardship-toolkit.git
cd data-stewardship-toolkit
pip install -e ".[dev]"
ds-steward check              # every licence / metadata / checksum rule
pytest -v                     # the same rules as a test suite
ds-steward check --json       # machine-readable report
# Ingest a file as a new version (checksum + provenance + inventory sync)
ds-steward ingest --dataset sample_air_quality \
    --file data/raw/air_quality_feb_2024.csv \
    --actor your.name@example.org --bump minor \
    --note "February 2024 batch"
# Register a published figure against a versioned input
ds-steward figure --figure-id fig02_pm25_trend \
    --dataset sample_air_quality --output figures/fig02.png \
    --script scripts/make_fig02.py --actor your.name@example.org
# Publish the repo to GitHub
git init && git add . && git commit -m "Data stewardship toolkit"
git remote add origin https://github.com/YOUR-USERNAME/data-stewardship-toolkit.git
git push -u origin main
